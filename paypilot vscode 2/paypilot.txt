const express = require('express');
const fs = require('fs');
const path = require('path');
require('dotenv').config();

const app = express();
const PORT = process.env.PORT || 3000;
const INVOICES_FILE = path.join(__dirname, 'invoices.json');

app.use(express.json());
app.use(express.static(path.join(__dirname, 'public')));

const PAYPAL_BASE = 'https://api-m.sandbox.paypal.com';
const PAYPAL_CLIENT_ID = process.env.PAYPAL_CLIENT_ID || '';
const PAYPAL_CLIENT_SECRET = process.env.PAYPAL_CLIENT_SECRET || '';
const LLM_API_KEY = process.env.LLM_API_KEY || process.env.GEMINI_API_KEY || '';

function readInvoices() {
  try {
    if (!fs.existsSync(INVOICES_FILE)) return [];
    return JSON.parse(fs.readFileSync(INVOICES_FILE, 'utf8') || '[]');
  } catch (err) {
    return [];
  }
}

function saveInvoices(invoices) {
  try {
    fs.writeFileSync(INVOICES_FILE, JSON.stringify(invoices, null, 2), 'utf8');
  } catch (err) {}
}

function isOverdue(inv) {
  if (inv.status === 'PAID' || !inv.due_date) return false;
  return new Date(inv.due_date + 'T23:59:59') < new Date();
}

function calculateCustomerRisks(invoices) {
  const customerOverdues = {};
  for (const inv of invoices) {
    const cust = inv.customer.trim();
    if (!customerOverdues[cust]) customerOverdues[cust] = 0;
    if (isOverdue(inv)) customerOverdues[cust]++;
  }
  const risks = {};
  for (const [cust, overdueCount] of Object.entries(customerOverdues)) {
    if (overdueCount === 0) risks[cust] = 'Low';
    else if (overdueCount === 1) risks[cust] = 'Medium';
    else risks[cust] = 'High';
  }
  return risks;
}
